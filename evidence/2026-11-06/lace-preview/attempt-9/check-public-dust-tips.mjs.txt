async function boundedBody(response) {
  if (!response.ok || !response.body) throw Error();
  const chunks = []; let length = 0;
  for await (const chunk of response.body) { length += chunk.length; if (length > 512 * 1024) throw Error(); chunks.push(chunk); }
  return JSON.parse(Buffer.concat(chunks).toString('utf8'));
}

    socket.on('open', () => socket.send(JSON.stringify({ type: 'connection_init' })));
    socket.on('error', () => finish({ status: 'transport-failed' }));
    socket.on('close', () => finish({ status: 'stream-closed' }));
    socket.on('message', bytes => { try {
      const message = JSON.parse(bytes.toString('utf8'));
      if (message.type === 'connection_ack') socket.send(JSON.stringify({ id: 'public-tip', type: 'subscribe', payload: { query, variables: { id: null } } }));
      else if (message.type === 'ping') socket.send(JSON.stringify({ type: 'pong' }));
      else if (message.type === 'error' || message.payload?.errors) finish({ status: 'query-rejected' });
      else if (message.type === 'next') {
        const value = message.payload?.data?.dustLedgerEvents;
        if (!Number.isSafeInteger(value?.id) || value.id < 0 || !Number.isSafeInteger(value.maxId) || value.maxId < value.id) return finish({ status: 'invalid-tip' });
        finish({ status: 'observed', id: value.id, maxId: value.maxId });
      }
    } catch { finish({ status: 'invalid-response' }); } });
  });
