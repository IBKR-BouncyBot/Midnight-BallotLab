function permittedColor(value) {
  for (const token of value.matchAll(/rgba?\(([^)]+)\)/g)) {
    const [r, g, b, a = 1] = token[1].split(/[,\s/]+/).map(Number);
    if (a === 0) continue;
    const neutral = r === g && g === b, blue = r === 0 && g === 0 && b >= 200;
    const green = g > r && g > b, red = r > g && r > b;
    if (!neutral && !blue && !green && !red) return false;
  }
  return true;
}
async function checkPalette(page) {
  // Compare the actual settled hover/focus/mode colors, not an interpolated
  // frame halfway through a neutral-to-blue CSS transition.
  await page.evaluate(async () => {
    const finite = document.getAnimations().filter(animation => animation.effect?.getComputedTiming().iterations !== Infinity);
    await Promise.all(finite.map(animation => animation.finished.catch(() => {})));
  });
  const entries = await page.evaluate(() => {
    const entries = [];
    for (const element of document.querySelectorAll('body, body *')) {
      const box = element.getBoundingClientRect(), style = getComputedStyle(element);
      if (!box.width || !box.height || style.visibility === 'hidden') continue;
      for (const pseudo of [null, '::before', '::after']) {
        const css = pseudo ? getComputedStyle(element, pseudo) : style;
        if (pseudo && ['none', 'normal'].includes(css.content)) continue;
        const properties = ['color', 'backgroundColor', 'backgroundImage', 'boxShadow', 'textShadow'];
        for (const side of ['Top', 'Right', 'Bottom', 'Left']) if (parseFloat(css[`border${side}Width`]) > 0) properties.push(`border${side}Color`);
        if (css.outlineStyle !== 'none' && parseFloat(css.outlineWidth) > 0) properties.push('outlineColor');
        for (const property of properties) entries.push({ element: element.id || `${element.tagName.toLowerCase()}.${String(element.className).replaceAll(' ', '.')}`,
          pseudo, property, value: css[property] });
      }
    }
    return entries;
  });
  const violations = entries.filter(entry => !permittedColor(entry.value));
  if (violations.length) report.paletteFailure = violations.slice(0, 12);
  assert.equal(violations.length, 0, 'Unexpected purple/tinted color in rendered theme');
  return [...new Set(entries.map(entry => entry.value))].sort();
}
