function safeUri(value, purpose) {
  const uri = new URL(value);
  assert.ok(!uri.username && !uri.password && !uri.search && !uri.hash, 'Expected credential-free service URI');
  if (purpose === 'prover') {
    assert.ok(['localhost', '127.0.0.1', '[::1]'].includes(uri.hostname) && uri.port === '6300', 'Expected local prover');
  } else {
    assert.ok(['https:', 'wss:'].includes(uri.protocol), 'Expected secure Preview service');
    assert.ok(['indexer.preview.midnight.network', 'rpc.preview.midnight.network'].includes(uri.hostname)
      || (uri.hostname === 'blockfrost.lw.iog.io' && uri.pathname.startsWith('/midnight-preview')),
    'Expected known Preview endpoint');
  }
  return uri.href;
}
