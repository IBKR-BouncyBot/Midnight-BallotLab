// One-off local layout/navigation check. No wallet, proofs or transactions.
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { createServer } from 'node:http';
import { readFile, mkdir, writeFile } from 'node:fs/promises';
import { createRequire } from 'node:module';
import { dirname, extname, join, resolve, sep } from 'node:path';
import { fileURLToPath } from 'node:url';
const root = resolve(dirname(fileURLToPath(import.meta.url)), '../..');
const dist = join(root, 'dist');
const output = join(root, 'evidence/2026-11-06/front-page-style');
process.env.PLAYWRIGHT_BROWSERS_PATH = join(root, '.browser-cache');
const requireContracts = createRequire(join(root, 'contracts/package.json'));
const hash = bytes => createHash('sha256').update(bytes).digest('hex');
const report = { executedAt: new Date().toISOString(), node: process.version,
  scope: 'local-landing-layout-and-real-destination-loaders', walletConnected: false,
  proofGeneration: false, networkSubmission: false, cases: [] };
let stage = 'build-prerequisites', server, browser;
const types = { '.html': 'text/html; charset=utf-8', '.css': 'text/css; charset=utf-8',
  '.mjs': 'text/javascript; charset=utf-8', '.js': 'text/javascript; charset=utf-8',
  '.json': 'application/json', '.wasm': 'application/wasm', '.svg': 'image/svg+xml',
  '.woff2': 'font/woff2' };
function isNeutralOrBlue(value) {
  for (const token of value.matchAll(/rgba?\(([^)]+)\)/g)) {
    const [r, g, b, a = 1] = token[1].split(/[,\s/]+/).map(Number);
    if (a === 0) continue;
    if (!(r === g && g === b) && !(r === 0 && g === 0 && b >= 200)) return false;
  }
  return true;
}
async function palette(page) {
  const values = await page.evaluate(() => {
    const result = new Set();
    for (const element of document.querySelectorAll('body, body *')) {
      const box = element.getBoundingClientRect(), style = getComputedStyle(element);
      if (!box.width || !box.height || style.visibility === 'hidden') continue;
      for (const pseudo of [null, '::before', '::after']) {
        const css = pseudo ? getComputedStyle(element, pseudo) : style;
        if (pseudo && ['none', 'normal'].includes(css.content)) continue;
        for (const prop of ['color', 'backgroundColor', 'backgroundImage', 'boxShadow', 'textShadow']) result.add(css[prop]);
        for (const side of ['Top', 'Right', 'Bottom', 'Left']) if (parseFloat(css[`border${side}Width`]) > 0) result.add(css[`border${side}Color`]);
        if (css.outlineStyle !== 'none' && parseFloat(css.outlineWidth) > 0) result.add(css.outlineColor);
      }
    }
    return [...result].sort();
  });
  assert.ok(values.every(isNeutralOrBlue), 'Rendered landing palette contains a non-neutral, non-blue color');
  return values;
}
try {
  await mkdir(output, { recursive: true });
  report.harnessSha256 = hash(await readFile(fileURLToPath(import.meta.url)));
  const native = JSON.parse(await readFile(join(dist, 'native-build.json'), 'utf8'));
  const network = JSON.parse(await readFile(join(dist, 'network-build.json'), 'utf8'));

  stage = 'chromium-launch';
  const { chromium } = requireContracts('playwright');
  report.playwright = requireContracts('playwright/package.json').version;
  browser = await chromium.launch({ headless: true }); report.chromium = browser.version();
  server = createServer(async (request, response) => {
    try {
      if (!['GET', 'HEAD'].includes(request.method)) { response.writeHead(405).end(); return; }
      let path = decodeURIComponent(new URL(request.url, 'http://127.0.0.1').pathname);
      if (path === '/favicon.ico') { response.writeHead(204).end(); return; }
      if (path.startsWith('/Midnight-BallotLab/')) path = path.slice('/Midnight-BallotLab'.length);
      if (path.endsWith('/')) path += 'index.html';
      const target = resolve(dist, `.${path}`);
      if (!target.startsWith(`${dist}${sep}`)) { response.writeHead(403).end(); return; }
      const bytes = await readFile(target);
      response.writeHead(200, { 'Content-Type': types[extname(target)] ?? 'application/octet-stream',
        'X-Content-Type-Options': 'nosniff', 'Cache-Control': 'no-store' });
      response.end(request.method === 'HEAD' ? undefined : bytes);
    } catch { response.writeHead(404).end(); }
  });
  await new Promise((yes, no) => { server.once('error', no); server.listen(0, '127.0.0.1', yes); });
  const origin = `http://127.0.0.1:${server.address().port}`;
  for (const prefix of ['/', '/Midnight-BallotLab/']) for (const width of [1440, 390, 320]) {
    const context = await browser.newContext({ viewport: { width, height: width === 1440 ? 1000 : 844 } });
    const result = { prefix, width, destinations: [] };
    try {
      for (const [demo, file, availability, assets] of [
        ['browser', 'native.html', 'native-availability', native.assets],
        ['testnet', 'network.html', 'network-availability', network.wasmAssets],
      ]) {
        stage = `${prefix}:${width}:${demo}:landing`;
        const page = await context.newPage(); page.setDefaultTimeout(60_000);
        let pageErrors = 0, consoleErrors = 0, sockets = 0;
        const requested = new Set(), loadedWasm = new Set();
        page.on('pageerror', () => { pageErrors++; });
        page.on('console', message => { if (message.type() === 'error') consoleErrors++; });
        page.on('request', request => { requested.add(request.url()); });
        page.on('websocket', () => { sockets++; });
        page.on('response', response => {
          if (response.ok() && new URL(response.url()).pathname.endsWith('.wasm')) loadedWasm.add(new URL(response.url()).pathname);
        });
        await page.goto(`${origin}${prefix}`, { waitUntil: 'networkidle' });
        assert.equal(await page.locator('#landing-title').count(), 1);
        assert.equal(await page.locator('[class*="orbit"], .landing-graphic, .landing-illustration').count(), 0);
        assert.equal(await page.locator('.landing-hero > *').count(), 1);
        assert.equal(await page.locator('.landing-hero svg, .landing-hero img, .landing-hero canvas').count(), 0);
        const layout = await page.evaluate(async () => {
          await document.fonts.ready;
          const hero = document.querySelector('.landing-hero'), intro = document.querySelector('.landing-introduction');
          return { viewport: document.documentElement.clientWidth,
            content: Math.max(document.documentElement.scrollWidth, document.body.scrollWidth),
            heroDisplay: getComputedStyle(hero).display, heroColumns: getComputedStyle(hero).gridTemplateColumns,
            heroWidth: hero.getBoundingClientRect().width, introductionWidth: intro.getBoundingClientRect().width,
            outfitLoaded: document.fonts.check('16px Outfit'), bodyBackground: getComputedStyle(document.body).backgroundColor };
        });
        assert.equal(layout.viewport, width); assert.ok(layout.content <= width + 1);
        assert.equal(layout.heroDisplay, 'block'); assert.equal(layout.heroColumns, 'none');
        assert.ok(layout.introductionWidth >= layout.heroWidth * 0.7); assert.equal(layout.outfitLoaded, true);
        assert.equal(layout.bodyBackground, 'rgb(10, 10, 10)');
        const colors = await palette(page);
        const cards = page.locator('#demo-options a[data-demo]'); assert.equal(await cards.count(), 2);
        for (const [target, href] of [['browser', './native.html'], ['testnet', './network.html']]) {
          const link = page.locator(`a[data-demo="${target}"]`);
          assert.equal(await link.getAttribute('href'), href); assert.equal(await link.isVisible(), true);
          assert.ok([null, '_self'].includes(await link.getAttribute('target')));
          assert.equal(await link.locator('.landing-card-action').evaluate(el => getComputedStyle(el).backgroundColor), 'rgb(0, 0, 254)');
        }
        assert.equal(loadedWasm.size, 0); assert.equal(sockets, 0);
        assert.equal(await page.evaluate(() => typeof globalThis.midnight), 'undefined');
        assert.deepEqual(await page.locator('script[src]').evaluateAll(elements => elements.map(el => el.getAttribute('src'))), ['./support.mjs']);
        assert.equal(await page.locator('link[rel="preload"], link[rel="modulepreload"], link[rel="prefetch"], link[rel="preconnect"]').count(), 0);
        for (const request of requested) {
          const url = new URL(request); assert.equal(url.origin, origin);
          assert.doesNotMatch(url.pathname, /(?:native|network)-(?:bundle|loader|assets|build)|\.wasm$/);
        }
        if (demo === 'browser') {
          // Full-page screenshots do not reliably trigger offscreen lazy images.

          await page.locator('.support-qr img').scrollIntoViewIfNeeded();
          await page.waitForFunction(() => {
            const image = document.querySelector('.support-qr img');
            return image.complete && image.naturalWidth > 0;
          });
          await page.evaluate(() => window.scrollTo(0, 0));
          const screenshot = `landing-${prefix === '/' ? 'root' : 'project'}-${width}.png`;
          await page.screenshot({ path: join(output, screenshot), fullPage: true, animations: 'disabled' });
          result.layout = layout; result.palette = colors; result.screenshot = screenshot; result.supportQrLoaded = true;
          if (width === 1440) {
            for (const target of ['browser', 'testnet']) {
              await page.locator(`a[data-demo="${target}"]`).hover(); await palette(page);
            }
          }
        }
        stage = `${prefix}:${width}:${demo}:actual-destination`;
        const link = page.locator(`a[data-demo="${demo}"]`);
        if (demo === 'browser') { await link.focus(); await link.press('Enter'); } else await link.click();
        await page.waitForURL(`${origin}${prefix}${file}`);
        await page.waitForFunction(id => ['ready', 'error', 'unavailable'].includes(document.getElementById(id)?.dataset.state), availability);
        assert.equal(await page.locator(`#${availability}`).getAttribute('data-state'), 'ready');
        for (const asset of assets) assert.ok(loadedWasm.has(`${prefix}${asset}`));
        if (demo === 'browser') {

          assert.equal(await page.locator('#native-create').isDisabled(), false);

          assert.equal(await page.locator('#network-deploy').isDisabled(), true);
          assert.equal(await page.locator('#network-events').textContent(), '');
        }
        for (const request of requested) assert.equal(new URL(request).origin, origin);
        assert.equal(pageErrors, 0); assert.equal(consoleErrors, 0); assert.equal(sockets, 0);
        result.destinations.push({ demo, ready: true, wasmLoaded: loadedWasm.size,
          activation: demo === 'browser' ? 'keyboard-enter' : 'pointer', runtimeBeforeChoice: false,
          externalRequests: 0, pageErrors, consoleErrors });
        await page.close();
      }
      result.passed = true; report.cases.push(result);
      console.log(JSON.stringify({ stage: `${prefix}:${width}`, passed: true }));
    } finally { await context.close(); }
  }
  report.passed = true; console.log(JSON.stringify({ landingStyle: 'passed', cases: report.cases.length }));
} catch (error) {
  report.passed = false; report.failure = { stage, category: error instanceof assert.AssertionError ? 'assertion' : 'browser-or-resource' };
  console.log(JSON.stringify(report.failure)); process.exitCode = 1;
} finally {
  await browser?.close(); if (server) await new Promise(resolveClose => server.close(resolveClose));
  await mkdir(output, { recursive: true }); await writeFile(join(output, 'landing-browser.json'), JSON.stringify(report, null, 2) + '\n');
}
