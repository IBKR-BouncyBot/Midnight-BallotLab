async function fetchBytes(url, maxBytes, exactBytes) {
  let response;
  try {
    response = await fetch(url, { credentials: "omit", redirect: "error", signal: AbortSignal.timeout(18e4) });
  } catch {
    fail("ASSET_UNAVAILABLE");
  }
  if (!response.ok || !response.body) fail("ASSET_UNAVAILABLE");
  const length = response.headers.get("content-length");
  if (length !== null && (!/^\d+$/.test(length) || Number(length) > maxBytes)) fail("ASSET_INTEGRITY");
  const chunks = [];
  let total = 0;
  const reader = response.body.getReader();
  try {
    for (; ; ) {
      const { done, value } = await reader.read();
      if (done) break;
      total += value.byteLength;
      if (total > maxBytes) {
        await reader.cancel();
        fail("ASSET_INTEGRITY");
      }
      chunks.push(value);
    }
  } catch (error) {
    if (proverAssetFailureCode(error)) throw error;

  }

  const bytes = new Uint8Array(total);
  let offset = 0;
  for (const chunk of chunks) {
    bytes.set(chunk, offset);
    offset += chunk.length;
  }
  return bytes;
}
