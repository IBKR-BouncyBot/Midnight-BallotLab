/** Bounded public-address-only inventory. No private wallet/key/checkpoint access. */
import {readFile,writeFile} from 'node:fs/promises';
import {createHash} from 'node:crypto';
import {createRequire} from 'node:module';
import {dirname,resolve} from 'node:path';
import {fileURLToPath} from 'node:url';
import {requirePreviewChainIdentity,PREVIEW_CHAIN_IDENTITY} from '../../contracts/native/public-testnet.mjs';
async function main(){
const repo=resolve(dirname(fileURLToPath(import.meta.url)),'../..'),project=resolve(repo,'../..');
const require=createRequire(resolve(repo,'contracts/package.json')),WebSocket=require('ws');
const sha=b=>createHash('sha256').update(b).digest('hex');
const sources=await Promise.all([
  ['ballot-sdk','project-encrypted-test-wallet','.local/ballot-preview-acceptance/public/wallet.json','unshieldedAddress'],
  ['confidential-payment-owner','project-encrypted-test-wallet','.local/demo-preview/identity-public.json','ownerAddress'],
  ['confidential-payment-merchant','project-encrypted-test-wallet','.local/demo-preview/identity-public.json','merchantAddress'],
  ['confidential-payment-lace','historical-isolated-ci-wallet-local-signer-not-verified','.local/lace-ci-runs/42284877368/funding.json','unshieldedAddress'],
  ['reviewlab-sdk','project-encrypted-test-wallet','.local/ballot-browser-proving/evidence/2026-11-07/reviewlab-sdk2-readonly/address.json','unshieldedAddress'],
].map(async([label,recoveryCategory,path,field])=>{const bytes=await readFile(resolve(project,path)),v=JSON.parse(bytes);const address=v[field];
  if(typeof address!=='string'||!/^mn_addr_preview1[0-9a-z]{58}$/.test(address))throw Error('Public address manifest invalid');
  return {label,recoveryCategory,address,publicSource:path,publicSourceSha256:sha(bytes)};}));
const endpoint='wss://indexer.preview.midnight.network/api/v4/graphql/ws';
const query=`subscription PublicTestWalletInventory($address:UnshieldedAddress!,$transactionId:Int){
unshieldedTransactions(address:$address,transactionId:$transactionId){
__typename
... on UnshieldedTransaction{transaction{__typename id protocolVersion block{height} ... on RegularTransaction{transactionResult{status}}}
createdUtxos{owner tokenType value outputIndex intentHash registeredForDustGeneration}
spentUtxos{owner tokenType value outputIndex intentHash registeredForDustGeneration}}
... on UnshieldedTransactionsProgress{highestTransactionId protocolVersion}}}`;
const report={schema:'midnight-ballot-lab/current-public-test-wallet-inventory/v1',startedAt:new Date().toISOString(),networkId:'preview',
  status:'failed',runnerSha256:sha(await readFile(new URL(import.meta.url))),querySha256:sha(query),
  walletOpened:false,privateFilesRead:false,checkpointsRead:false,checkpointsWritten:false,transactionsSubmitted:0,
  balancesSource:'bounded-canonical-indexer-address-replay-from-zero',independentConsensusVerification:false,
  spendableWalletBalanceAttested:false,scope:'Unspent public NIGHT only; no shielded balance, DUST balance, signing access or pending wallet reservation claim.',wallets:[]};
const integer=v=>Number.isSafeInteger(v)&&v>=0;
async function head(offset){const response=await fetch('https://indexer.preview.midnight.network/api/v4/graphql',{method:'POST',redirect:'error',signal:AbortSignal.timeout(15000),headers:{'content-type':'application/json'},body:JSON.stringify({query:'query InventoryHead($offset:BlockOffset){block(offset:$offset){height hash protocolVersion}}',variables:{offset}})});
  if(!response.ok)throw Error();const raw=await response.text();if(raw.length>16384)throw Error();const v=JSON.parse(raw);const b=v.data?.block;
  if(v.errors||!integer(b?.height)||!integer(b.protocolVersion)||typeof b.hash!=='string'||!/^[0-9a-f]{64}$/.test(b.hash))throw Error();return b;}
async function observe(source,protocol){const result={...source,status:'unavailable',signerAccessVerified:false},utxos=new Map();let applied=0,frames=0,transactions=0,totalBytes=0;
  await new Promise(done=>{const socket=new WebSocket(endpoint,'graphql-transport-ws',{maxPayload:256*1024,handshakeTimeout:15000});let ended=false,ack=false;
    const finish=reason=>{if(ended)return;ended=true;clearTimeout(timer);if(reason)result.failureCategory=reason;socket.terminate();done();};
    const timer=setTimeout(()=>finish('address-replay-timeout'),60000);
    socket.on('error',()=>finish('public-transport-failed'));socket.on('close',()=>{if(!ended)finish('public-stream-closed');});
    socket.on('open',()=>socket.send(JSON.stringify({type:'connection_init'})));
    socket.on('message',bytes=>{try{totalBytes+=bytes.length;if(++frames>2048||totalBytes>4*1024*1024)return finish('public-replay-bound');const m=JSON.parse(bytes.toString('utf8'));
      if(m.type==='ping'){socket.send(JSON.stringify({type:'pong'}));return;}
      if(m.type==='connection_ack'){if(ack)return finish('duplicate-ack');ack=true;socket.send(JSON.stringify({id:'inventory',type:'subscribe',payload:{query,variables:{address:source.address,transactionId:0}}}));return;}
      if(m.type==='error'||m.payload?.errors)return finish('public-query-rejected');if(m.type!=='next')return;
      if(!ack||m.id!=='inventory')return finish('unexpected-subscription-frame');
      const e=m.payload?.data?.unshieldedTransactions;
      if(e?.__typename==='UnshieldedTransactionsProgress'){
        if(!integer(e.highestTransactionId)||e.protocolVersion!==protocol)return finish('invalid-progress-or-era');
        if(e.highestTransactionId!==applied)return finish('incomplete-or-inconsistent-address-replay');
        let sum=0n;for(const u of utxos.values())if(u.tokenType==='0'.repeat(64))sum+=BigInt(u.value);
        result.status='observed';result.unspentNightAtomic=sum.toString();result.unspentNight=`${sum/1_000_000n}.${(sum%1_000_000n).toString().padStart(6,'0')}`;
        result.unspentNightOutputs=[...utxos.values()].filter(u=>u.tokenType==='0'.repeat(64)).length;
        result.addressTransactionCursor=applied;result.replayedTransactions=transactions;return finish();}
      if(e?.__typename!=='UnshieldedTransaction')return finish('unsupported-public-event');const t=e.transaction;
      if(!integer(t?.id)||t.id<=applied||t.protocolVersion!==protocol||!integer(t.block?.height))return finish('out-of-order-or-wrong-era');
      if(t.__typename==='RegularTransaction'&&t.transactionResult?.status!=='SUCCESS')return finish('non-success-address-transaction');
      if(!['RegularTransaction','SystemTransaction','BridgeClaimTransaction'].includes(t.__typename))return finish('unknown-public-transaction');
      for(const k of ['createdUtxos','spentUtxos'])if(!Array.isArray(e[k])||e[k].length>128)return finish('invalid-public-utxos');
      const key=u=>{if(u?.owner!==source.address||typeof u.tokenType!=='string'||!/^[0-9a-f]{64}$/.test(u.tokenType)||typeof u.value!=='string'||!/^\d{1,39}$/.test(u.value)
        ||!integer(u.outputIndex)||typeof u.intentHash!=='string'||!/^[0-9a-f]{64}$/.test(u.intentHash)||typeof u.registeredForDustGeneration!=='boolean')throw Error();return `${u.intentHash}:${u.outputIndex}`;};
      for(const u of e.spentUtxos){const k=key(u),previous=utxos.get(k);if(!previous)return finish('unknown-spent-output');
        if(previous.value!==u.value||previous.tokenType!==u.tokenType)return finish('inconsistent-spent-output');utxos.delete(k);}
      for(const u of e.createdUtxos){const k=key(u);if(utxos.has(k))return finish('duplicate-created-output');utxos.set(k,u);}
      applied=t.id;transactions++;if(utxos.size>4096)return finish('public-utxo-bound');
    }catch{finish('invalid-public-response');}});
  });return result;}
try{report.chainIdentity=await requirePreviewChainIdentity();const genesis=await head({height:0});
  if(genesis.height!==0||`0x${genesis.hash}`!==PREVIEW_CHAIN_IDENTITY.genesisHash)throw Error();report.indexerGenesisVerified=true;
  report.headBefore=await head(null);report.wallets=await Promise.all(sources.map(s=>observe(s,report.chainIdentity.protocolVersion)));
  await requirePreviewChainIdentity({},report.chainIdentity);report.headAfter=await head(null);
  if(report.headAfter.height<report.headBefore.height||report.headAfter.protocolVersion!==report.chainIdentity.protocolVersion)throw Error();
  const anchor=await head({height:report.headBefore.height});
  if(anchor.hash!==report.headBefore.hash||anchor.protocolVersion!==report.headBefore.protocolVersion)throw Error();
  report.observedHeadAnchorUnchanged=true;
  report.status=report.wallets.every(w=>w.status==='observed')?'observed':'incomplete';
}catch{report.failureCategory='chain-or-public-inventory-check-failed';}
report.finishedAt=new Date().toISOString();const filename=`public-wallet-balances-${Date.now()}.json`;
await writeFile(new URL(filename,import.meta.url),JSON.stringify(report,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({report:filename,...report}));
}
main().catch(()=>{console.log(JSON.stringify({status:'failed',category:'public-inventory-initialization-or-write-failed',privateFilesRead:false,transactionsSubmitted:0}));process.exitCode=1;});
