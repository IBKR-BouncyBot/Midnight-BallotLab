async function waitUntil(seconds, label) {
  await setStage(label);
  // Require both real wall time and the actual indexed block timestamp.
  // Neither the browser clock nor the contract's sealed deadline is changed.
  const deadline = Math.max(Date.now(), (seconds + 6) * 1000) + 300_000;
  for (;;) {
    if (Date.now() >= (seconds + 6) * 1000) {
      const response = await fetch('http://127.0.0.1:8088/api/v4/graphql', {
        method: 'POST', headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ query: '{ block { timestamp } }' }), signal: AbortSignal.timeout(5000) });
      assert.ok(response.ok);
      const value = await response.json(), timestamp = value?.data?.block?.timestamp;
      assert.ok(!value.errors && Number.isSafeInteger(timestamp));
      if (timestamp >= seconds * 1000) return;
    }
    if (Date.now() >= deadline) throw new Error('bounded-stage-timeout');
    await delay(1000);
  }
}

    if (command.trim() === 'stop') { await stop(); lines.close(); return; }

  // A detached stdin must not discard private state before root review.
  await new Promise(() => {});
}
